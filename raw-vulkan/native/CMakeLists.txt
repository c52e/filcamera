cmake_minimum_required(VERSION 3.22)
project(rawvulkan LANGUAGES C CXX)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../gpu-runtime" gpu-runtime)
add_library(jpeg-static STATIC IMPORTED)
set_target_properties(jpeg-static PROPERTIES IMPORTED_LOCATION "${JPEG_BUILD_DIR}/libjpeg.a")
add_library(rawvulkan SHARED raw.cpp dng_jpeg.cpp)
target_compile_features(rawvulkan PRIVATE cxx_std_17)
target_compile_options(rawvulkan PRIVATE -O2 -Wall -Wextra -ffp-contract=off)
target_include_directories(rawvulkan PRIVATE "${RAW_SHADER_DIR}" "${JPEG_SOURCE_DIR}/src" "${JPEG_BUILD_DIR}")
target_link_libraries(rawvulkan PRIVATE vulkan jpeg-static filmcamera_gpu)
target_link_options(rawvulkan PRIVATE -Wl,-z,max-page-size=16384)
